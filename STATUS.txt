cd /workspaces/IOT-Home-Security/esp32cam-test

mkdir -p firmware-release

# Copy compiled firmware binaries
cp .pio/build/esp32cam/bootloader.bin firmware-release/
cp .pio/build/esp32cam/partitions.bin firmware-release/
cp .pio/build/esp32cam/firmware.bin firmware-release/

# Copy boot_app0.bin from PlatformIO's installed ESP32 Arduino framework
BOOT_APP0="$(find ~/.platformio/packages -name boot_app0.bin -type f 2>/dev/null | head -n 1)"

if [ -n "$BOOT_APP0" ]; then
    cp "$BOOT_APP0" firmware-release/
    echo "boot_app0.bin found: $BOOT_APP0"
else
    echo "WARNING: boot_app0.bin was not found."
fi

# Create ZIP containing the available binaries
cd firmware-release
zip -j ../DHARSSHITH-ESP32CAM-firmware7.zip \
    bootloader.bin partitions.bin firmware.bin \
    $( [ -f boot_app0.bin ] && echo boot_app0.bin )

cd ..

ls -lh firmware-release/
